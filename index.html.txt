<html lang="zh-TW">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>PDF 批次改名工具</title>
    <script src="https://cdn.tailwindcss.com"></script>
    <script src="https://cdnjs.cloudflare.com/ajax/libs/jszip/3.10.1/jszip.min.js"></script>
    <script src="https://cdnjs.cloudflare.com/ajax/libs/xlsx/0.18.5/xlsx.full.min.js"></script>
    <script src="https://unpkg.com/lucide@latest"></script>
    <script src="https://unpkg.com/react@18/umd/react.production.min.js"></script>
    <script src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
</head>
<body class="min-h-screen bg-slate-50 text-slate-800 font-sans pb-12">
    <div id="root"></div>
    <script type="text/babel">
        const { useState, useRef, useCallback } = React;
        const { UploadCloud, FileText, Download, Trash2, Settings, Archive, Table, GripVertical, FileEdit, X } = lucide;

        function App() {
            const [files, setFiles] = useState([]);
            const [excelStartRow, setExcelStartRow] = useState(2);
            const fileInputRef = useRef(null);
            const excelInputRef = useRef(null);

            const handleFiles = (newFiles) => {
                const pdfs = Array.from(newFiles).filter(f => f.name.toLowerCase().endsWith('.pdf'));
                const mappedPdfs = pdfs.map(f => ({
                    id: crypto.randomUUID(), file: f, newBaseName: f.name.replace(/\.[^/.]+$/, ""), selected: true
                }));
                setFiles(prev => [...prev, ...mappedPdfs]);
            };

            const handleExcelUpload = async (e) => {
                const file = e.target.files[0];
                if (!file) return;
                const reader = new FileReader();
                reader.onload = (evt) => {
                    const data = new Uint8Array(evt.target.result);
                    const workbook = XLSX.read(data, { type: 'array' });
                    const worksheet = workbook.Sheets[workbook.SheetNames[0]];
                    const range = XLSX.utils.decode_range(worksheet['!ref']);
                    const dataRows = [];
                    for (let R = excelStartRow - 1; R <= range.e.r; R++) {
                        let rowData = [];
                        for (let C = range.s.c; C <= range.e.c; C++) {
                            const cell = worksheet[XLSX.utils.encode_cell({ c: C, r: R })];
                            if (cell) rowData.push(String(cell.w || cell.v).trim());
                        }
                        if (rowData.length > 0) dataRows.push(rowData.join('_'));
                    }
                    setFiles(prev => prev.map((f, i) => i < dataRows.length ? {...f, newBaseName: dataRows[i]} : f));
                };
                reader.readAsArrayBuffer(file);
            };

            const downloadAll = async () => {
                const zip = new JSZip();
                files.filter(f => f.selected).forEach(f => zip.file(`${f.newBaseName}.pdf`, f.file));
                const content = await zip.generateAsync({type: 'blob'});
                const a = document.createElement('a');
                a.href = URL.createObjectURL(content);
                a.download = 'Renamed_PDFs.zip';
                a.click();
            };

            return (
                <div className="max-w-6xl mx-auto p-6 space-y-6">
                    <h1 className="text-2xl font-bold">PDF 批次改名工具</h1>
                    <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
                        <div className="space-y-4">
                            <button onClick={() => fileInputRef.current.click()} className="w-full bg-blue-600 text-white p-4 rounded-lg">上傳 PDF</button>
                            <input type="file" multiple className="hidden" ref={fileInputRef} onChange={(e) => handleFiles(e.target.files)} />
                            <div className="bg-white p-4 rounded shadow border">
                                <label>Excel 起始行:</label>
                                <input type="number" value={excelStartRow} onChange={e=>setExcelStartRow(parseInt(e.target.value))} className="w-full border p-2 mb-2"/>
                                <button onClick={() => excelInputRef.current.click()} className="w-full bg-green-600 text-white p-2 rounded">匯入 Excel</button>
                                <input type="file" className="hidden" ref={excelInputRef} onChange={handleExcelUpload} />
                            </div>
                        </div>
                        <div className="lg:col-span-2 bg-white p-6 rounded shadow border">
                            {files.map((f, i) => (
                                <div key={f.id} className="flex items-center gap-2 mb-2 p-2 bg-slate-50 border rounded">
                                    <input type="checkbox" checked={f.selected} onChange={() => setFiles(prev => prev.map(item => item.id === f.id ? {...item, selected: !item.selected} : item))} />
                                    <input value={f.newBaseName} onChange={(e) => setFiles(prev => prev.map(item => item.id === f.id ? {...item, newBaseName: e.target.value} : item))} className="flex-1 p-1 border rounded"/>
                                </div>
                            ))}
                            <button onClick={downloadAll} className="w-full mt-4 bg-black text-white py-3 rounded font-bold">打包下載</button>
                        </div>
                    </div>
                </div>
            );
        }
        ReactDOM.createRoot(document.getElementById('root')).render(<App />);
    </script>
</body>
</html>
